CREATE TABLE DISPOSITIVI(
    Codice_Dispositivo INTEGER,
    CONSTRAINT PK_DISPOSITIVI PRIMARY KEY (Codice_Dispositivo)
);

CREATE TABLE CASELLI(
    Numero_Casello INTEGER,
    Nome_Uscita VARCHAR(30),
    Id_Casello INTEGER,
    CONSTRAINT PK_CASELLI PRIMARY KEY (Id_Casello)
);  

CREATE TABLE CLIENTI(
    Nome VARCHAR(20) NOT NULL,
    Cognome VARCHAR(20) NOT NULL,
    Data_Nascita DATE NOT NULL,
    Id_Cliente INTEGER,
    Via VARCHAR(15),
    Citta VARCHAR(15),
    Civico INTEGER,
    CAP INTEGER,
    CONSTRAINT PK_CLIENTI PRIMARY KEY (Id_Cliente)
);

CREATE TABLE AUTOMOBILI(
    Targa VARCHAR(7),
    Modello VARCHAR(30),
    Cliente INTEGER NOT NULL,
    Proprieta VARCHAR(2) CHECK(Proprieta = 'si' OR Proprieta = 'no'),
    Codice_Dispositivo INTEGER NOT NULL,
    CONSTRAINT PK_AUTOMOBILI PRIMARY KEY (Targa)
);

CREATE TABLE CARTE_DI_CREDITO(
    Id_Carta INTEGER,
    Numero_Carta VARCHAR(20) NOT NULL,
    Scadenza DATE NOT NULL,
    Id_Cliente INTEGER,
    CONSTRAINT PK_C_D_C PRIMARY KEY (Id_Carta)
);  

CREATE TABLE CONTI_CORRENTI(
    Id_Conto INTEGER,
    Numero_Conto VARCHAR(20) NOT NULL,
    Id_Cliente INTEGER,
	CONSTRAINT PK_C_C PRIMARY KEY (Id_Conto)
);

CREATE TABLE TRAGITTI(
    Numero_Tragitto INTEGER,
    Codice_Dispositivo INTEGER NOT NULL,
    Data_In TIMESTAMP,
    Data_Out TIMESTAMP,
    Numero_Casello_In INTEGER,
    Numero_Casello_Out INTEGER,
	CONSTRAINT PK_TRAGITTI PRIMARY KEY (Numero_Tragitto, Codice_Dispositivo)
);

--Verifichiamo che la data di uscita sia successiva a quella di ingresso
CREATE OR REPLACE TRIGGER DATE_TRAGITTO_CONSISTENTI
BEFORE INSERT OR UPDATE ON TRAGITTI
FOR EACH ROW
DECLARE
	ERRORE EXCEPTION;
BEGIN
    IF :NEW.Data_Out <= :NEW.Data_In THEN
    	RAISE ERRORE;
    END IF;
EXCEPTION
    	WHEN ERRORE THEN
			RAISE_APPLICATION_ERROR(-20001, 
    		'La data di uscita deve essere successiva alla data di ingresso');
END;

--Verifichiamo che per ogni dispositivo ci siano al più 2 auto di un unico cliente
CREATE OR REPLACE TRIGGER NUMERO_AUTO
BEFORE INSERT OR UPDATE ON AUTOMOBILI
FOR EACH ROW
DECLARE
    NUMERO_AUTO INTEGER;
	ALTRO_CLIENTE INTEGER;
	ERRORE_NUM_AUTO EXCEPTION;
    ERRORE_CLIENTI_DIVERSI EXCEPTION;
BEGIN
    --Controlliamo il numero di auto già associate al dispositivo
    SELECT COUNT(*) INTO NUMERO_AUTO FROM AUTOMOBILI
    WHERE Codice_Dispositivo = :NEW.Codice_Dispositivo; --AND CLIENTE = NEW:CLIENTE;
    
    IF NUMERO_AUTO >= 2 THEN
        RAISE ERRORE_NUM_AUTO;
    END IF;

	--Controlliamo che il dispositivo sia utilizzato da un unico cliente
	SELECT COUNT(*) INTO ALTRO_CLIENTE
    FROM AUTOMOBILI
    WHERE Codice_Dispositivo = :NEW.Codice_Dispositivo AND Cliente <> :NEW.Cliente;

    IF ALTRO_CLIENTE > 0 THEN --(Il dispositivo è già associato a un altro cliente)
        RAISE ERRORE_CLIENTI_DIVERSI;
    END IF;

EXCEPTION
    WHEN ERRORE_NUM_AUTO THEN
        RAISE_APPLICATION_ERROR(-20002, 'Sono già presenti 2 automobili associate allo stesso dispositivo');
	WHEN ERRORE_CLIENTI_DIVERSI THEN
        RAISE_APPLICATION_ERROR(-20002, 'Il dispositivo è di un altro cliente');
END;

/

--Verifichiamo che esista un metodo di pagamento quando si inserisce un'automobile
CREATE OR REPLACE TRIGGER METODO_PAGAMENTO_ESISTENTE
BEFORE INSERT OR UPDATE ON AUTOMOBILI
FOR EACH ROW
DECLARE
    NUM_CONTI INTEGER;
	NUM_CARTE INTEGER;
	ERRORE EXCEPTION;
BEGIN
    SELECT COUNT (*) INTO NUM_CARTE
    FROM CARTE_DI_CREDITO
    WHERE Id_Cliente = :NEW.Cliente;

    SELECT COUNT (*) INTO NUM_CONTI
    FROM CONTI_CORRENTI
    WHERE Id_Cliente = :NEW.Cliente;

    IF (NUM_CARTE = 0 AND NUM_CONTI = 0) THEN
    	RAISE ERRORE;
	END IF;
EXCEPTION
    WHEN ERRORE THEN
    	RAISE_APPLICATION_ERROR(-20005, 
    		'Il cliente associato all automobile appena inserita non possiede 
   			 alcun metodo di pagamento!');
END;

--Verifichiamo che il cliente non inserisca più di una carta
CREATE OR REPLACE TRIGGER LIMITE_CARTE
BEFORE INSERT ON CARTE_DI_CREDITO
FOR EACH ROW
DECLARE
    NUMERO_CARTE INTEGER;
	ERRORE EXCEPTION;
BEGIN
    SELECT COUNT(CDC.Id_Carta) INTO NUMERO_CARTE
    FROM CARTE_DI_CREDITO CDC JOIN CLIENTI C ON C.Id_Cliente = CDC.Id_Cliente
    WHERE C.Id_Cliente = :NEW.Id_Cliente;

    IF NUMERO_CARTE > 0 THEN
        RAISE ERRORE;
    END IF;
EXCEPTION
    WHEN ERRORE THEN
        RAISE_APPLICATION_ERROR(-20006, 'Il cliente ha già una carta inserita');
END;

--Verifichiamo che il cliente non inserisca più di un conto
CREATE OR REPLACE TRIGGER LIMITE_CONTI
BEFORE INSERT ON CONTI_CORRENTI
FOR EACH ROW
DECLARE
    NUMERO_CONTI INTEGER;
	ERRORE EXCEPTION;
BEGIN
    SELECT COUNT(CC.Id_Conto) INTO NUMERO_CONTI
    FROM CONTI_CORRENTI CC JOIN CLIENTI C ON C.Id_Cliente = CC.Id_Cliente
    WHERE C.Id_Cliente = :NEW.Id_Cliente;

    IF NUMERO_CONTI > 0 THEN
        RAISE ERRORE;
    END IF;
EXCEPTION
    WHEN ERRORE THEN
        RAISE_APPLICATION_ERROR(-20007, 'Il cliente ha già un conto inserito');
END;

CREATE VIEW TRAGITTI_PER_CLIENTE
AS SELECT COUNT(T.Numero_Tragitto) AS NUM_TRAGITTI, C.Nome AS NOME, 
C.Cognome AS COGNOME, C.Id_Cliente AS CODICE_CLIENTE
FROM (CLIENTI C LEFT JOIN AUTOMOBILI A ON C.Id_Cliente = A.Cliente)
LEFT JOIN TRAGITTI T ON T.Codice_Dispositivo = A.Codice_Dispositivo
GROUP BY C.Id_Cliente, C.Nome, C.Cognome;

--STORED PROCEDURE PER IL NUMERO DI TRAGITTI PER CLIENTE
CREATE OR REPLACE PROCEDURE TRAGITTI_TOTALI(
    Cliente_Input IN CLIENTI.Id_Cliente%TYPE,
    Num_Tragitti_Output OUT INTEGER
) IS
BEGIN
	SELECT NUM_TRAGITTI INTO Num_Tragitti_Output
	FROM TRAGITTI_PER_CLIENTE --Vista
	WHERE CODICE_CLIENTE = Cliente_Input;

	DBMS_OUTPUT.PUT_LINE('Il cliente con ID = ' || Cliente_Input || ' ha effettuato ' 
    || Num_Tragitti_Output || ' tragitti.');

END;

--Vista che ci serve per la prossima stored Procedure
CREATE VIEW TRAGITTI_NON_CONCLUSI
AS SELECT COUNT(T.Numero_Tragitto) AS NUM_TRAGITTI, T.Numero_Casello_In, C.Nome_Uscita AS NOME
FROM TRAGITTI T JOIN CASELLI C ON T.Numero_Casello_In = C.Id_Casello
WHERE T.Numero_Casello_Out IS NULL
GROUP BY T.Numero_Casello_In, C.Nome_Uscita;

--Stored Procedure per la segnalazione dell'uscita più trafficata
CREATE OR REPLACE PROCEDURE USCITA_PIU_TRAFFICATA(
    USCITA_TRAFFICATA OUT CASELLI.Nome_Uscita%TYPE
) IS
BEGIN
	SELECT NOME INTO USCITA_TRAFFICATA
	FROM TRAGITTI_NON_CONCLUSI
	WHERE NUM_TRAGITTI = (SELECT MAX(NUM_TRAGITTI) FROM TRAGITTI_NON_CONCLUSI);

	DBMS_OUTPUT.PUT_LINE('. Si sconsiglia di NON prendere l uscita ' || USCITA_TRAFFICATA || 
    ' in quanto è la più trafficata!');

END;

CREATE OR REPLACE PROCEDURE Inserimento_Automobile(
    p_Targa IN AUTOMOBILI.Targa%TYPE,
    p_Modello IN AUTOMOBILI.Modello%TYPE,
    p_Cliente IN AUTOMOBILI.Cliente%TYPE,
    p_Proprieta IN AUTOMOBILI.Proprieta%TYPE, 
    p_CodiceDispositivo IN AUTOMOBILI.Codice_Dispositivo%TYPE
)
IS
BEGIN
    INSERT INTO AUTOMOBILI(Targa, Modello, Cliente, Proprieta, Codice_Dispositivo) 
    VALUES (p_Targa, p_Modello, p_Cliente, p_Proprieta, p_CodiceDispositivo);
    COMMIT;

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Errore durante inserimento automobile');
        ROLLBACK;
        RAISE;
END Inserimento_Automobile;
    
/

--Esempio 
DECLARE
    v_Targa              AUTOMOBILI.Targa%TYPE := 'CZ963ZZ';
    v_Modello            AUTOMOBILI.Modello%TYPE := 'Chevrolet Matiz';
    v_Cliente            AUTOMOBILI.Cliente%TYPE := 2;
    v_Proprieta          AUTOMOBILI.Proprieta%TYPE := 'si';
    v_CodiceDispositivo  AUTOMOBILI.Codice_Dispositivo%TYPE := 5;

BEGIN
    Inserimento_Automobile(v_Targa, v_Modello, v_Cliente, v_Proprieta, v_CodiceDispositivo);

    DBMS_OUTPUT.PUT_LINE('Automobile inserita con successo');

END;
